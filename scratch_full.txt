const MAIN_SHEET = "AJIO_RETURN";

// ── NEW column layout (1-based) ──────────────────────────────
const COL_SELLER_SKU = 3;
const COL_RETURN_QTY = 4;
const COL_CUST_ORDER = 5;
const COL_RETURN_CREATED = 6;   // F  
const COL_RETURN_DELIVERED = 8;   // H  
const COL_RETURN_AWB = 9;   // I  
const COL_ACTUAL_DELIVERED = 10;   // J  manual
const COL_QUALITY = 11;   // K  manual
const COL_CARRIER = 13;
const COL_RETURN_ORDER_NUM = 15;
const TOTAL_COLS = 15;

const PURPLE = "#9C2BE6";   // Alert 1 — Delivered Not Received
const ORANGE = "#FF9900";   // Alert 2 — 61+ Days No Actual Delivery

// ================================================================
// ENTRY POINTS
// ================================================================

function doPost(e) {
  try {
    runAlerts();
    return ContentService.createTextOutput("OK");
  } catch (err) {
    Logger.log("doPost error: " + err.message);
    return ContentService.createTextOutput("ERROR: " + err.message);
  }
}

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("📋 AJIO Alerts")
    .addItem("▶  Run Alerts / Check Statuses", "runAlerts")
    .addSeparator()
    .addItem("🎨  Refresh All Highlights", "refreshAllHighlights")
    .addSeparator()
    .addItem("🔄  Sync Returns to RTO", "syncDailyReturnsToRto")
    .addToUi();
}

// ================================================================
// MAIN ALERT LOGIC
// ================================================================

function runAlerts() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const mainWs = ss.getSheetByName(MAIN_SHEET);

  if (!mainWs) {
    Logger.log("ERROR: Required sheet not found");
    return;
  }

  const dataRows = mainWs.getDataRange().getValues().slice(1);
  updateCancelledReturns(mainWs, dataRows);
  refreshAllHighlights();

  Logger.log("runAlerts complete ✅");
}

function updateCancelledReturns(mainWs, dataRows) {
  const updates = [];
  for (let i = 0; i < dataRows.length; i++) {
    const qty = String(dataRows[i][COL_RETURN_QTY - 1] ?? "").trim();
    const quality = String(dataRows[i][COL_QUALITY - 1] || "").trim();

    if (qty === "0" && quality !== "Return Cancelled") {
      updates.push({ row: i + 2, col: COL_QUALITY, val: "Return Cancelled" });
      dataRows[i][COL_QUALITY - 1] = "Return Cancelled"; // update in memory so downstream skips it
    } else if (qty !== "0" && quality === "Return Cancelled") {
      // Revert back to Select if QTY is no longer 0
      updates.push({ row: i + 2, col: COL_QUALITY, val: "Select" });
      dataRows[i][COL_QUALITY - 1] = "Select";
    }
  }
  if (updates.length > 0) {
    updates.forEach(u => mainWs.getRange(u.row, u.col).setValue(u.val));
    Logger.log("Updated " + updates.length + " Quality statuses.");
  }
}

// ================================================================
// onEdit — instant highlight cleanup
// ================================================================

function onEdit(e) {
  if (!e || !e.range) return;
  const sh = e.range.getSheet();
  if (sh.getName() !== MAIN_SHEET) return;

  const editedCols = [];
  for (let j = 0; j < e.range.getNumColumns(); j++) {
    editedCols.push(e.range.getColumn() + j);
  }

  // If either Actual Delivered Date (J) or Quality (K) is edited, refresh highlights
  if (editedCols.includes(COL_ACTUAL_DELIVERED) || editedCols.includes(COL_QUALITY)) {
    refreshAllHighlights();
  }
}

// ================================================================
// HIGHLIGHTS (Single Source of Truth)
// ================================================================

function refreshAllHighlights() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const mainWs = ss.getSheetByName(MAIN_SHEET);
  if (!mainWs) return;

  const lastRow = mainWs.getLastRow();
  if (lastRow < 2) return;

  // Single batch read
  const allData = mainWs.getRange(2, 1, lastRow - 1, TOTAL_COLS).getValues();
  const colours = [];

  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const wEnd = new Date(today);
  wEnd.setDate(today.getDate() - 3);

  const cutoff61 = new Date(today);
  cutoff61.setDate(today.getDate() - 61);

  for (let i = 0; i < allData.length; i++) {
    const row = allData[i];
    const ron = String(row[COL_RETURN_ORDER_NUM - 1] || "").trim();
    const actualVal = String(row[COL_ACTUAL_DELIVERED - 1] || "").trim();
    const qty = String(row[COL_RETURN_QTY - 1] ?? "").trim();
    const quality = String(row[COL_QUALITY - 1] || "").trim();

    let colour = null; // default: clear

    // Only consider highlighting if RON exists, QTY != 0, and item is NOT physically received
    if (ron && qty !== "0" && actualVal === "") {
      const delivDate = parseDate(row[COL_RETURN_DELIVERED - 1]);
      const createdDt = parseDate(row[COL_RETURN_CREATED - 1]);

      // Check Purple Condition
      if (delivDate && delivDate <= wEnd && quality !== "Non Delivered 7 days") {
        colour = PURPLE;
      }
      // Check Orange Condition (Only if H is empty)
      else if (!delivDate && createdDt && createdDt < cutoff61 && quality !== "Non Delivered 60+ days") {
        colour = ORANGE;
      }
    }

    colours.push(Array(TOTAL_COLS).fill(colour));
  }

  // Single batch write
  mainWs.getRange(2, 1, allData.length, TOTAL_COLS).setBackgrounds(colours);
  Logger.log(`refreshAllHighlights done ✅ — ${allData.length} rows processed`);
}

// ================================================================
// DATE PARSER
// ================================================================

function parseDate(val) {
  if (!val) return null;
  if (val instanceof Date && !isNaN(val)) {
    const d = new Date(val); d.setHours(0, 0, 0, 0); return d;
  }
  const s = String(val).trim();
  if (!s) return null;
  // dd-mm-yyyy HH:MM  or  dd-mm-yyyy
  const m1 = s.match(/^(\d{2})-(\d{2})-(\d{4})/);
  if (m1) { const d = new Date(+m1[3], +m1[2] - 1, +m1[1]); d.setHours(0, 0, 0, 0); return d; }
  // yyyy-mm-dd
  const m2 = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (m2) { const d = new Date(+m2[1], +m2[2] - 1, +m2[3]); d.setHours(0, 0, 0, 0); return d; }
  const d = new Date(s);
  return isNaN(d) ? null : d;
}

// Hardcoded IDs (Replace with your actual values if they change)
const MASTER_INVENTORY_ID = '1GMfVnYMuXxB-3_m1kuPpBcnuKK6zO02mcg0hmPprAn4';
const ALERT_EMAIL = 'whitehuskfashion@gmail.com,care.whitehusk@gmail.com'; // REPLACE THIS WITH ACTUAL EMAIL

function cleanSku(skuStr) {
  if (!skuStr) return "";
  // Strip "SKU:" (case insensitive) and surrounding spaces/quotes
  let s = String(skuStr).replace(/['"]/g, '').replace(/sku:/i, '').trim();
  return s;
}

function cleanSpaces(sku) {
  return String(sku).replace(/\s+/g, '').toUpperCase();
}

function syncDailyReturnsToRto() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  // 1. Connect to Master Inventory and Load Mapping Sheet
  let masterSS;
  try {
    masterSS = SpreadsheetApp.openById(MASTER_INVENTORY_ID);
  } catch (e) {
    SpreadsheetApp.getUi().alert("Error: Could not open Print Number and Qty spreadsheet. Check ID and permissions.");
    return;
  }

  const mappingWs = masterSS.getSheetByName("Mapping Sheet");
  if (!mappingWs) {
    SpreadsheetApp.getUi().alert("Error: 'Mapping Sheet' not found in Print Number and Qty spreadsheet.");
    return;
  }

  // Load Mapping into Space-Insensitive Dictionary
  const mappingData = mappingWs.getDataRange().getValues();
  const spaceInsensitiveMapping = {};

  // Loop through rows (skip header row 0)
  for (let i = 1; i < mappingData.length; i++) {
    const row = mappingData[i];
    // SKUs start from Column D (index 3)
    for (let j = 3; j < row.length; j++) {
      const validSku = String(row[j] || "").trim();
      if (validSku) {
        spaceInsensitiveMapping[cleanSpaces(validSku)] = validSku;
      }
    }
  }

  // Define tabs to process
  const tabs = ['website_return', 'myntra_return', 'AJIO_RETURN', 'FLIPKART_RETURN', 'pop_return'];

  const validSkusToAdd = {}; // { "WHF-OS-SUKUNA-M": 3 }
  const invalidSkus = [];    // Array of objects for email report

  let stats = {
    evaluated: 0,
    synced: 0,
    badCondition: 0,
    incorrectSku: 0,
    dateMissing: 0,
    waitAndSee: 0
  };

  for (let t = 0; t < tabs.length; t++) {
    const sheetName = tabs[t];
    const ws = ss.getSheetByName(sheetName);
    if (!ws) continue;

    const lastRow = ws.getLastRow();
    if (lastRow < 2) continue;

    const lastCol = ws.getLastColumn();
    const headers = ws.getRange(1, 1, 1, lastCol).getValues()[0];

    // Dynamically find columns (1-indexed for SpreadsheetApp, 0-indexed for array)
    const findCol = (nameArr) => {
      for (let i = 0; i < headers.length; i++) {
        const h = String(headers[i]).trim().toLowerCase();
        if (nameArr.some(n => h === n.toLowerCase())) return i;
      }
      return -1;
    };

    // Ensure SYNC_STATUS column exists
    let syncColIdx = findCol(["SYNC_STATUS", "SYNC STATUS"]);
    if (syncColIdx === -1) {
      ws.getRange(1, lastCol + 1).setValue("SYNC_STATUS");
      syncColIdx = lastCol; // 0-indexed index of new column
    }

    const conditionColIdx = findCol(["Condition", "Quality"]);
    const skuColIdx = findCol(["Products", "SELLER SKU"]);
    const dateColIdx = findCol(["Date", "Actual Delivered Date"]);
    const qtyColIdx = findCol(["Return QTY", "QTY"]); // Specifically for AJIO
    const orderIdColIdx = findCol(["Order ID", "Cust Order No", "Store Package Id", "Order id"]); // Dynamic Order ID

    if (conditionColIdx === -1 || skuColIdx === -1) {
      Logger.log(`Skipping ${sheetName} - Could not find Condition or SKU columns.`);
      continue;
    }

    const rangeToUpdate = ws.getRange(2, 1, lastRow - 1, Math.max(lastCol, syncColIdx + 1));
    const allData = rangeToUpdate.getValues();
    const allBackgrounds = rangeToUpdate.getBackgrounds();
    const updates = []; // Array to batch update SYNC_STATUS
    let hasBgUpdates = false;

    for (let r = 0; r < allData.length; r++) {
      const row = allData[r];
      const syncStatus = String(row[syncColIdx] || "").trim().toUpperCase();
      const condition = String(row[conditionColIdx] || "").trim().toLowerCase();

      // Skip if already successfully synced or explicitly marked as bad condition
      if (syncStatus === "SYNCED" || syncStatus === "BAD_CONDITION") continue;



      // Extract SKUs (handling comma separated strings)
      const rawSkuCell = String(row[skuColIdx] || "").trim();
      
      // Check if Date is filled
      let hasDate = false;
      let rowDate = "No Date Provided";
      if (dateColIdx !== -1 && row[dateColIdx]) {
        const rawDate = row[dateColIdx];
        if (rawDate instanceof Date) {
          rowDate = rawDate.toLocaleDateString();
          hasDate = true;
        } else {
          rowDate = String(rawDate).trim();
          if (rowDate !== "") hasDate = true;
        }
      }

      // If SKU is completely empty, it's a blank row. Reset status if it was previously an error, then skip.
      if (!rawSkuCell) {
        if (syncStatus === "INCORRECT_SKU" || syncStatus === "DATE_MISSING") {
          updates.push({ row: r + 2, col: syncColIdx + 1, val: "" });
          for (let c = 0; c < allBackgrounds[r].length; c++) {
             allBackgrounds[r][c] = "#ffffff";
          }
          hasBgUpdates = true;
        }
        continue;
      }
      
      const conditionStr = String(row[conditionColIdx] || "").trim().toLowerCase();
      
      // A) Wait and See Conditions (Matches "Return Cancelled", "Non Delivered")
      if (conditionStr.includes("cancelled") || conditionStr.includes("non delivered")) {
        // We do nothing for now. But if it previously had an error, wipe it clean.
        if (syncStatus === "INCORRECT_SKU" || syncStatus === "DATE_MISSING") {
          updates.push({ row: r + 2, col: syncColIdx + 1, val: "" }); // Reset status
          for (let c = 0; c < allBackgrounds[r].length; c++) {
             allBackgrounds[r][c] = "#ffffff"; // Reset color
          }
          hasBgUpdates = true;
        stats.waitAndSee++;
          stats.evaluated++;
        }
        continue;
      }
      
      // B) Date Check for everything else EXCEPT "Select"
      if (!hasDate) {
         if (conditionStr.includes("select")) {
            // It's just an uninspected, untouched row. Wipe errors if any, then ignore.
            if (syncStatus === "INCORRECT_SKU" || syncStatus === "DATE_MISSING") {
              updates.push({ row: r + 2, col: syncColIdx + 1, val: "" }); 
              for (let c = 0; c < allBackgrounds[r].length; c++) { 
                 allBackgrounds[r][c] = "#ffffff"; 
              }
              hasBgUpdates = true;
            stats.waitAndSee++;
              stats.evaluated++;
            }
            continue;
         } else {
            // It's Good or Bad, but missing a date!
            updates.push({ row: r + 2, col: syncColIdx + 1, val: "DATE_MISSING" });
            for (let c = 0; c < allBackgrounds[r].length; c++) { 
               allBackgrounds[r][c] = "#ffff00"; // Yellow
            }
            hasBgUpdates = true;
            stats.dateMissing++;
            stats.evaluated++;
            continue;
         }
      }
      
      // C) Terminal Bad Conditions (Matches "Bad", "Bad - Ticket needed", "Torn", etc.)
      if (conditionStr.includes("bad") || conditionStr.includes("torn") || conditionStr.includes("damage") || conditionStr.includes("fail") || conditionStr.includes("wrong")) {
        updates.push({ row: r + 2, col: syncColIdx + 1, val: "BAD_CONDITION" });
        if (syncStatus === "INCORRECT_SKU" || syncStatus === "DATE_MISSING") {
          for (let c = 0; c < allBackgrounds[r].length; c++) {
             allBackgrounds[r][c] = "#ffffff";
          }
          hasBgUpdates = true;
        }
        stats.badCondition++;
        stats.evaluated++;
        continue;
      }
      
      // D) Process as Good
      let skuList = rawSkuCell.split(',').map(s => cleanSku(s)).filter(s => s.length > 0);
      
      // Handle AJIO QTY Logic
      let multiplier = 1;
      if (sheetName === 'AJIO_RETURN' && qtyColIdx !== -1) {
        const qtyVal = parseInt(row[qtyColIdx], 10);
        if (isNaN(qtyVal) || qtyVal <= 0) continue; // Skip if 0 or invalid QTY
        multiplier = qtyVal;
      }
      
      // Get Order ID for email reporting
      let rowOrderId = "N/A";
      if (orderIdColIdx !== -1 && row[orderIdColIdx]) {
        rowOrderId = String(row[orderIdColIdx]).trim() || "N/A";
      }

      let allValid = true;
      let rowValidSkus = {}; // Temporary holding area for this row's valid SKUs
      
      // Validate each SKU in the cell
      for (let i = 0; i < skuList.length; i++) {
        const cleanedSku = skuList[i];
        const spacelessSku = cleanSpaces(cleanedSku);
        
        const exactlyMatchedSku = spaceInsensitiveMapping[spacelessSku];
        
        if (exactlyMatchedSku) {
          rowValidSkus[exactlyMatchedSku] = (rowValidSkus[exactlyMatchedSku] || 0) + multiplier;
        } else {
          allValid = false;
          // Add to invalid array (skip if empty)
          if (cleanedSku) {
            invalidSkus.push({ sheet: sheetName, sku: cleanedSku, date: rowDate, orderId: rowOrderId });
          }
        }
      }
      
      if (allValid) {
        // Only if EVERY SKU in the row is valid, we add them to the master list
        for (let sku in rowValidSkus) {
           validSkusToAdd[sku] = (validSkusToAdd[sku] || 0) + rowValidSkus[sku];
        }
        updates.push({ row: r + 2, col: syncColIdx + 1, val: "SYNCED" });
        
        // If this row previously had a typo (was marked INCORRECT_SKU or DATE_MISSING), clear the yellow highlight!
        if (syncStatus === "INCORRECT_SKU" || syncStatus === "DATE_MISSING") {
          for (let c = 0; c < allBackgrounds[r].length; c++) {
             allBackgrounds[r][c] = "#ffffff";
          }
          hasBgUpdates = true;
        }
        stats.synced++;
        stats.evaluated++;
      } else {
        // Mark as INCORRECT_SKU (will be retried next time)
        updates.push({ row: r + 2, col: syncColIdx + 1, val: "INCORRECT_SKU" });
        // Highlight entire row in yellow
        for (let c = 0; c < allBackgrounds[r].length; c++) {
           allBackgrounds[r][c] = "#ffff00";
        }
        hasBgUpdates = true;
      stats.incorrectSku++;
        stats.evaluated++;
      }
    }
    
    // Batch write SYNC_STATUS updates
    if (updates.length > 0) {
      updates.forEach(u => ws.getRange(u.row, u.col).setValue(u.val));
    }

    // Batch write row color updates (Extremely fast batch operation)
    if (hasBgUpdates) {
      rangeToUpdate.setBackgrounds(allBackgrounds);
    }
  }

  // 3. Write Valid SKUs to rto_inventory
  const rtoKeys = Object.keys(validSkusToAdd);
  if (rtoKeys.length > 0) {
    const rtoWs = masterSS.getSheetByName("rto_inventory");
    if (!rtoWs) {
      SpreadsheetApp.getUi().alert("Error: 'rto_inventory' tab not found in destination spreadsheet.");
      return;
    }

    const rtoLastRow = Math.max(1, rtoWs.getLastRow());
    const rtoData = rtoWs.getRange(1, 1, rtoLastRow, 2).getValues();

    // Create a map of existing rows in RTO
    const existingRto = {}; // { "WHF-OS-M": 5 } -> Row 5
    for (let r = 1; r < rtoData.length; r++) {
      const sku = String(rtoData[r][0]).trim().toUpperCase();
      if (sku) existingRto[sku] = r + 1; // 1-indexed row
    }

    const newRowsToAppend = [];

    for (let i = 0; i < rtoKeys.length; i++) {
      const sku = rtoKeys[i];
      const searchSku = sku.toUpperCase();
      const qtyToAdd = validSkusToAdd[sku];

      if (existingRto[searchSku]) {
        // Update existing row
        const rowNum = existingRto[searchSku];
        const currentCount = parseInt(rtoWs.getRange(rowNum, 2).getValue(), 10) || 0;
        rtoWs.getRange(rowNum, 2).setValue(currentCount + qtyToAdd);
      } else {
        // Append new row
        newRowsToAppend.push([
          sku,
          qtyToAdd,
          0, // Locked
          `=B${rtoLastRow + 1 + newRowsToAppend.length}-C${rtoLastRow + 1 + newRowsToAppend.length}` // OnHand Formula
        ]);
      }
    }

    // Batch append new rows
    if (newRowsToAppend.length > 0) {
      rtoWs.getRange(rtoLastRow + 1, 1, newRowsToAppend.length, 4).setValues(newRowsToAppend);
    }
  }

  // 4. Send Email if there are invalid SKUs
  if (invalidSkus.length > 0) {
    let htmlBody = `
      <h3>Invalid SKUs Detected During RTO Sync</h3>
      <p>The following SKUs were marked as 'Good' condition but could not be found in the Mapping Sheet. They have <b>NOT</b> been added to the rto_inventory.</p>
      <table border="1" cellpadding="8" cellspacing="0" style="border-collapse: collapse; text-align: left;">
        <tr style="background-color: #f2f2f2;">
          <th>Sheet Source</th>
          <th>Invalid SKU</th>
          <th>Order ID</th>
          <th>Row Date</th>
        </tr>
    `;

    for (let i = 0; i < invalidSkus.length; i++) {
      htmlBody += `
        <tr>
          <td>${invalidSkus[i].sheet}</td>
          <td style="color: red; font-family: monospace;">${invalidSkus[i].sku}</td>
          <td style="font-family: monospace;">${invalidSkus[i].orderId}</td>
          <td>${invalidSkus[i].date}</td>
        </tr>
      `;
    }

    htmlBody += `</table><br><p>Please correct these entries in your spreadsheets and ensure they are added to the mapping sheet if they are valid products.</p>`;

    MailApp.sendEmail({
      to: ALERT_EMAIL,
      subject: `🚨 Action Required: ${invalidSkus.length} Invalid RTO SKUs Found`,
      htmlBody: htmlBody
    });
  }

  let popupMessage = `✅ Sync Complete!\n\n` +
    `📊 Breakdown of Newly Evaluated Rows: ${stats.evaluated}\n` +
    `--------------------------------------------------\n` +
    `🟢 Successfully Synced to RTO: ${stats.synced} row(s)\n` +
    `🔴 Marked as Bad Condition: ${stats.badCondition} row(s)\n` +
    `🟡 Invalid SKUs (Yellow): ${stats.incorrectSku} row(s)\n` +
    `🟡 Missing Dates (Yellow): ${stats.dateMissing} row(s)\n` +
    `⏳ Waiting on Staff/Courier: ${stats.waitAndSee} row(s)\n\n` +
    `Total Unique SKUs Updated in Master Inventory: ${rtoKeys.length}`;
    
  SpreadsheetApp.getUi().alert(popupMessage);
}
